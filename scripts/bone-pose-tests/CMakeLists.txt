cmake_minimum_required(VERSION 3.20)
project(bone_pose_tests LANGUAGES CXX)
enable_testing()
find_package(OpenSceneGraph 3.6 REQUIRED osg osgAnimation osgUtil)
add_executable(bone-pose-tests test.cpp)
target_compile_features(bone-pose-tests PRIVATE cxx_std_17)
target_include_directories(bone-pose-tests PRIVATE ../.. ${OPENSCENEGRAPH_INCLUDE_DIRS})
target_link_libraries(bone-pose-tests PRIVATE ${OPENSCENEGRAPH_LIBRARIES})
add_test(NAME bone-pose COMMAND bone-pose-tests)
